```
        candidate = item.data(Qt.ItemDataRole.UserRole)
        return candidate if isinstance(candidate, TitleCandidate) else None

    def _apply_state(self, state: TitleState) -> None:
        candidate = self._selected_candidate()
        if candidate is None or self._service is None or not self._project_key:
            return
        if state is not TitleState.PROPOSED and candidate.state is state:
            return
        if state is TitleState.PROPOSED and candidate.state is TitleState.PROPOSED:
            return
        try:
            self._service.set_state(
                self._project_key, _row_id(candidate), state
            )
        except TitleBuilderNotFoundError as exc:
            self._show_error("Candidate not found", str(exc))
            return
        except TitleBuilderError as exc:
            self._show_error("Unable to update the candidate", str(exc))
            return
        self.refresh()

    def _update_actions(self, candidate: TitleCandidate | None) -> None:
        has_selection = candidate is not None
        state = candidate.state if candidate is not None else None
        self._accept_button.setEnabled(
            has_selection and state is not TitleState.ACCEPTED
        )
        self._reject_button.setEnabled(
            has_selection and state is not TitleState.REJECTED
        )
        self._archive_button.setEnabled(
            has_selection and state is not TitleState.ARCHIVED
        )
        self._restore_button.setEnabled(
            has_selection and state is not TitleState.PROPOSED
        )
        self._edit_button.setEnabled(has_selection)
        self._use_button.setEnabled(has_selection)

    def _show_error(self, title: str, message: str) -> None:
        DialogFactory.message(self, title, message, kind=MessageKind.ERROR)


def _row_id(candidate: TitleCandidate) -> int:
    identifier = candidate.identifier
    if identifier.startswith("row."):
        try:
            return int(identifier.split(".", 1)[1])
        except (ValueError, IndexError):
            return 0
    return 0
```

```
                "covers the essentials."

            return
```

```
"""Search orchestration service.

The orchestrator executes a provider-neutral search request across the
selected sources, aggregates the results, deduplicates the merged hits,
and â€” when requested â€” persists the outcome through the repository
layer.

Design decisions
----------------
- Adapters are executed on a bounded thread pool. The pool size is
  driven by the request's ``max_concurrency``. Each adapter runs on its
  own thread; the orchestrator does not share mutable state across
  threads beyond the future collection.
- Failures are isolated per source. A failure in one adapter never
  aborts the whole search; the outcome records the failure and the
  remaining sources proceed.
- The orchestrator is the only place that composes the source registry,
  the provider registry, the publication service, and the repository
  layer. Callers supply a ``RepositoryManager``, a
  ``SearchSourceService``, and a ``ProviderRegistry``; the orchestrator
  obtains everything else from those.
- The orchestrator never commits. It runs inside a unit of work
  supplied by the caller.
- When persistence is requested, the orchestrator records one
  ``SearchSession`` row, one ``SearchResult`` row per deduplicated
  publication, and one ``Publication`` row per publication. Session and
  result rows are recorded only after the search has completed, so a
  failure during search never leaves a partial session in the database.
"""

class SearchOrchestrator:
    """Execute searches across multiple sources and aggregate the results."""

    def __init__(
        self,
        repositories: RepositoryManager,
        source_service: SearchSourceService,
        provider_registry: ProviderRegistry,
    ) -> None:
        self._repositories = repositories
        self._source_service = source_service
        self._provider_registry = provider_registry
        self._publications = PublicationService(repositories)

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------
    def execute(
        self,
        request: SearchOrchestrationRequest,
        *,
        saved_query_id: int | None = None,
    ) -> SearchOrchestrationResponse:
        """Execute a search request across the selected sources.

        Parameters
        ----------
        request:
            The orchestration request.
        saved_query_id:
            Optional identifier of the saved search query this execution
            corresponds to. Recorded on the ``SearchSession`` row when
            persistence is requested.
        """
        sources = self._resolve_sources(request)
        if not sources:
            raise SearchOrchestrationConfigurationError(
                "No enabled sources matched the selection"
            )

        started = time.monotonic()
        hits_by_source, outcomes = self._execute_all(sources, request)
        hits = _merge_hits_in_order(sources, hits_by_source)

        if not request.record_session:
            deduplication = _empty_deduplication(hits)
            return SearchOrchestrationResponse(
                query=request.search_request.query,
                scope=request.session_scope,
                outcomes=tuple(outcomes),
                hits=tuple(hits),
                publications=deduplication.publications,
                deduplication=deduplication,
                session_id=None,
                result_ids=(),
                partial=_is_partial(outcomes),
            )

        session_id, result_ids, publications = self._record_outcome(
            request=request,
            sources=sources,
            hits=hits,
            outcomes=outcomes,
            saved_query_id=saved_query_id,
        )

        deduplication = _empty_deduplication(hits)
        deduplication = type(deduplication)(
            publications=publications,
            merges=deduplication.merges,
            input_count=deduplication.input_count,
        )

        _logger.info(
            "Search completed: %d sources, %d hits, %d unique publications, %.2fs",
            len(sources),
            len(hits),
            len(publications),
            time.monotonic() - started,
        )

        return SearchOrchestrationResponse(
            query=request.search_request.query,
            scope=request.session_scope,
            outcomes=tuple(outcomes),
            hits=tuple(hits),
            publications=publications,
            deduplication=deduplication,
            session_id=session_id,
            result_ids=tuple(result_ids),
            partial=_is_partial(outcomes),
        )

    # ------------------------------------------------------------------
    # Source resolution
    # ------------------------------------------------------------------
    def _resolve_sources(
        self,
        request: SearchOrchestrationRequest,
    ) -> list[SourceRecord]:
        if request.selection.all_enabled:
            return list(self._source_service.list_enabled())
        resolved: list[SourceRecord] = []
        for key in request.selection.source_keys:
            try:
                record = self._source_service.get_by_key(key)
            except SourceNotFoundError:
                _logger.warning("Source %s was selected but is not configured", key)
                continue
            if not record.config.enabled:
                _logger.info("Source %s was selected but is disabled", key)
                continue
            resolved.append(record)
        return resolved

    # ------------------------------------------------------------------
    # Execution
    # ------------------------------------------------------------------
    def _execute_all(
        self,
        sources: list[SourceRecord],
        request: SearchOrchestrationRequest,
    ) -> tuple[dict[str, list[SearchHit]], list[SourceOutcome]]:
        hits_by_source: dict[str, list[SearchHit]] = {}
        outcomes: list[SourceOutcome] = []

        with ThreadPoolExecutor(max_workers=request.max_concurrency) as executor:
            futures: dict[Future, SourceRecord] = {}
            for source in sources:
                future = executor.submit(self._execute_source, source, request)
                futures[future] = source

            for future in as_completed(futures):
                source = futures[future]
                outcome, hits = self._collect_future(source, future)
                outcomes.append(outcome)
                hits_by_source[source.key] = hits

        outcomes.sort(key=lambda o: o.display_name.lower())
        return hits_by_source, outcomes

    def _execute_source(
        self,
        source: SourceRecord,
        request: SearchOrchestrationRequest,
    ) -> tuple[SourceOutcome, list[SearchHit]]:
        started = time.monotonic()
        try:
            adapter = self._source_service.resolver.resolve(source.config)
        except SourceError as exc:
            return (
                SourceOutcome(
                    source_key=source.key,
                    provider_key=source.config.provider_key,
                    display_name=source.config.display_name,
                    status=SourceOutcomeStatus.FAILED,
                    message=f"Unable to resolve provider: {exc}",
                    duration_ms=0,
                ),
                [],
            )

        provider_request = _bind_source_to_request(request.search_request, source)
        try:
            response: SearchResponse = adapter.search(provider_request)
        except ProviderError as exc:
            return (
                SourceOutcome(
                    source_key=source.key,
                    provider_key=source.config.provider_key,
                    display_name=source.config.display_name,
                    status=SourceOutcomeStatus.FAILED,
                    message=str(exc),
                    duration_ms=int((time.monotonic() - started) * 1000),
                ),
                [],
            )
        except Exception as exc:  # noqa: BLE001 - defensive boundary
            _logger.exception(
                "Adapter %s raised an unexpected exception", source.key
            )
            return (
                SourceOutcome(
                    source_key=source.key,
                    provider_key=source.config.provider_key,
                    display_name=source.config.display_name,
                    status=SourceOutcomeStatus.FAILED,
                    message=f"Unexpected error: {exc}",
                    duration_ms=int((time.monotonic() - started) * 1000),
                ),
                [],
            )

        status_report = response.status or SourceStatusReport(
            provider_key=source.config.provider_key,
            status=SourceStatus.OK,
        )
        hits = list(response.hits)
        outcome_status = _translate_status(status_report.status)
        return (
            SourceOutcome(
                source_key=source.key,
                provider_key=source.config.provider_key,
                display_name=source.config.display_name,
                status=outcome_status,
                result_count=len(hits),
                duration_ms=status_report.duration_ms
                if status_report.duration_ms is not None
                else int((time.monotonic() - started) * 1000),
                message=status_report.message,
            ),
            hits,
        )

    def _collect_future(
        self,
        source: SourceRecord,
        future: Future,
    ) -> tuple[SourceOutcome, list[SearchHit]]:
        try:
            return future.result()
        except Exception as exc:  # noqa: BLE001 - defensive boundary
            _logger.exception(
                "Source %s raised during execution", source.key
            )
            return (
                SourceOutcome(
                    source_key=source.key,
                    provider_key=source.config.provider_key,
                    display_name=source.config.display_name,
                    status=SourceOutcomeStatus.FAILED,
                    message=f"Unexpected error: {exc}",
                    duration_ms=None,
                ),
                [],
            )

    # ------------------------------------------------------------------
    # Persistence
    # ------------------------------------------------------------------
    def _record_outcome(
        self,
        *,
        request: SearchOrchestrationRequest,
        sources: list[SourceRecord],
        hits: list[SearchHit],
        outcomes: list[SourceOutcome],
        saved_query_id: int | None,
    ) -> tuple[int, list[int], tuple[CanonicalPublication, ...]]:
        import datetime as dt
        import json

        session_repo = self._repositories.get("search_session")
        result_repo = self._repositories.get("search_result")

        partial = _is_partial(outcomes)
        source_reports = [
            {
                "source_key": outcome.source_key,
                "provider_key": outcome.provider_key,
                "display_name": outcome.display_name,
                "status": outcome.status.value,
                "result_count": outcome.result_count,
                "duration_ms": outcome.duration_ms,
                "message": outcome.message,
            }
            for outcome in outcomes
        ]

        session_row = SearchSession(
            query_id=saved_query_id,
            query_text=request.search_request.query,
            scope=request.session_scope,
            executed_at=dt.datetime.now(dt.timezone.utc).replace(tzinfo=None),
            source_reports_json=json.dumps(source_reports),
            partial=1 if partial else 0,
            result_count=len(hits),
            message="",
        )
        try:
            session_repo.add(session_row)
        except RepositoryError as exc:
            raise SearchOrchestrationError(
                f"Unable to record search session: {exc}"
            ) from exc

        deduplication = self._deduplicate_hits(hits)
        try:
            publications = self._publications.persist(deduplication.publications)
        except RepositoryError as exc:
            raise SearchOrchestrationError(
                f"Unable to persist publications: {exc}"
            ) from exc

        publication_by_key = {
            publication.key: row for publication, row in zip(
                deduplication.publications, publications
            )
        }

        result_ids: list[int] = []
        for rank, publication in enumerate(deduplication.publications):
            row = publication_by_key.get(publication.key)
            if row is None:
                continue
            try:
                result = result_repo.add_result(
                    session_id=session_row.id,
                    publication_id=row.id,
                    provider_key=publication.source_providers[0]
                    if publication.source_providers
                    else "unknown",
                    rank=rank,
                    score=None,
                    source_url=publication.url,
                )
            except RepositoryError as exc:
                _logger.warning(
                    "Unable to record search result for %s: %s",
                    publication.key,
                    exc,
                )
                continue
            result_ids.append(result.id)

        return session_row.id, result_ids, deduplication.publications

    def _deduplicate_hits(self, hits: list[SearchHit]):
        from academic_platform.application.publications.deduplicator import (
            PublicationDeduplicator,
        )
        from academic_platform.application.publications.normalizer import (
            PublicationNormalizer,
        )

        normalizer = PublicationNormalizer()
        normalization = normalizer.normalize(hits)
        deduplicator = PublicationDeduplicator()
        return deduplicator.deduplicate(list(normalization.publications))

def _translate_status(status: SourceStatus) -> SourceOutcomeStatus:
    if status is SourceStatus.OK:
        return SourceOutcomeStatus.OK
    if status is SourceStatus.FAILED:
        return SourceOutcomeStatus.FAILED
    if status is SourceStatus.DISABLED:
        return SourceOutcomeStatus.DISABLED
    return SourceOutcomeStatus.SKIPPED


def _merge_hits_in_order(
    sources: list[SourceRecord],
    hits_by_source: dict[str, list[SearchHit]],
) -> list[SearchHit]:
    ordered: list[SearchHit] = []
    for source in sources:
        ordered.extend(hits_by_source.get(source.key, []))
    return ordered


def _is_partial(outcomes: list[SourceOutcome]) -> bool:
    has_ok = any(o.status is SourceOutcomeStatus.OK for o in outcomes)
    has_failure = any(o.status is SourceOutcomeStatus.FAILED for o in outcomes)
    return has_ok and has_failure


def _empty_deduplication(hits: list[SearchHit]):
    from academic_platform.application.publications.deduplicator import (
        DeduplicationResult,

    return DeduplicationResult(
        publications=(),
        merges=(),
        input_count=len(hits),

def build_request_from_saved_search(
    saved: SavedSearch,
    *,
    limit: int = 50,
    session_scope: str | None = None,
) -> SearchOrchestrationRequest:
    """Construct an orchestration request from a saved search.

    The saved search's ``source_keys`` are honoured when present;
    otherwise every enabled source is targeted. The saved search's
    ``query_text`` becomes the free-text query, and its filter payload is
    used to populate the provider-neutral ``SearchRequest`` fields that
    the saved search supports.
    """
    if not saved.query_text.strip():
        raise ValueError("saved search has no query text")
    filters = saved.filters or {}
    search_request = _SearchRequest(
        query=saved.query_text,
        limit=limit,
        offset=int(filters.get("offset", 0) or 0),
        year_from=_optional_int(filters.get("year_from")),
        year_to=_optional_int(filters.get("year_to")),
        author=_optional_str(filters.get("author")),
        venue=_optional_str(filters.get("venue")),
        doi=_optional_str(filters.get("doi")),
        open_access_only=bool(filters.get("open_access_only", False)),
        language=_optional_str(filters.get("language")),
        sort=_sort_from_filters(filters),
    )
    if saved.source_keys:
        selection = _SourceSelection(
            all_enabled=False,
            source_keys=tuple(saved.source_keys),
        )
    else:
        selection = _SourceSelection(all_enabled=True)
    scope = session_scope or saved.scope or "remote"
    if scope not in {"local", "remote", "all"}:
        scope = "remote"
    return SearchOrchestrationRequest(
        search_request=search_request,
        selection=selection,
        session_scope=scope,
    )


def _optional_int(value) -> int | None:
    if value is None:
        return None
    try:
        return int(value)
    except (TypeError, ValueError):
        return None


def _optional_str(value) -> str | None:
    if value is None:
        return None
    text = str(value).strip()
    return text or None


def _sort_from_filters(filters: dict) -> str:
    value = filters.get("sort")
    if isinstance(value, str) and value in {"relevance", "date_desc", "date_asc"}:
        return value
    return "relevance"
```

```
"""Immutable value objects describing a search orchestration request.

The orchestration request combines the provider-neutral ``SearchRequest``
with the set of sources to target and the concurrency policy. It is
constructed by the presentation layer or by the external API boundary
and is passed unchanged to the orchestrator.

Design decisions
----------------
- The source selection is a set of source keys. The orchestrator
  resolves the keys to source records through the source registry
  service. This keeps the request decoupled from the source registry's
  internal identifiers.
- ``all_enabled`` selects every source that the source registry reports
  as enabled. When ``all_enabled`` is True, the explicit ``source_keys``
  is ignored; this makes the caller's intent unambiguous.
- The concurrency policy is bounded. A value below 1 or above the
  application's maximum is rejected at construction.
"""

from __future__ import annotations

from dataclasses import dataclass, field

from academic_platform.infrastructure.search_providers.adapter import (
    SearchRequest,
)

_MAX_CONCURRENCY = 16


@dataclass(frozen=True, slots=True)
class SourceSelection:
    """Describes which sources a search targets.

    Attributes
    ----------
    all_enabled:
        When True, every enabled source is targeted. The explicit
        ``source_keys`` is ignored.
    source_keys:
        The source keys to target when ``all_enabled`` is False.
    """

    all_enabled: bool = True
    source_keys: tuple[str, ...] = field(default_factory=tuple)

    def __post_init__(self) -> None:
        if not self.all_enabled and not self.source_keys:
            raise ValueError(
                "source_keys must not be empty when all_enabled is False"
            )


@dataclass(frozen=True, slots=True)
class SearchOrchestrationRequest:
    """A search request across one or more sources.

    Attributes
    ----------
    search_request:
        The provider-neutral search request executed against each
        selected source.
    selection:
        The set of sources to target.
    max_concurrency:
        Maximum number of adapters executed in parallel. Bounded to
        ``_MAX_CONCURRENCY``.
    record_session:
```

### FILE: src\academic_platform\application\search_sources\catalog_installer.py
```
"""Install the built-in source catalog into the database.

The installer is idempotent. It creates missing built-in sources and
refreshes the display-level fields of existing ones. It never removes a
built-in source, and it never modifies a user-defined source.

Design decisions
----------------
- Installation runs during application startup, after migrations and
  after settings reconciliation. A failure is reported and does not
  prevent the UI from starting; the search page falls back to the null
  adapter.
- Existing built-in sources keep their user-editable fields (enabled,
  result_limit, timeout_seconds). Only the display-level fields that the
  application ships (display_name, base_url, endpoint, parameter_map,
  capabilities) are refreshed. This is a deliberate choice: the user
  can change the enabled state and the limits of a built-in source
  without losing those changes on the next application start.
"""

from __future__ import annotations

import json
import logging

from academic_platform.application.search_sources.catalog import (
    BUILTIN_SOURCES,
    BuiltinSource,
)
from academic_platform.infrastructure.repositories.manager import (
    RepositoryManager,
)
from academic_platform.infrastructure.repositories.errors import (
    RepositoryError,
)

_logger = logging.getLogger(__name__)


class CatalogInstaller:
    """Install built-in sources into the database."""

    def __init__(self, repositories: RepositoryManager) -> None:
        self._repositories = repositories

    def install(self) -> tuple[int, int]:
        """Install the catalog.

        Returns ``(created, refreshed)``. A source that already exists is
        refreshed only when its display-level fields differ from the
        catalog.
        """
        created = 0
        refreshed = 0
        for entry in BUILTIN_SOURCES:
            outcome = self._install_entry(entry)
            if outcome == "created":
                created += 1
            elif outcome == "refreshed":
                refreshed += 1
        return created, refreshed

    def _install_entry(self, entry: BuiltinSource) -> str:
        repo = self._repositories.get("search_source")
        existing = repo.find_by_key(entry.config.key)
        if existing is None:
            repo.create(_create_payload(entry))
            _logger.info("Installed built-in source %s", entry.config.key)
            return "created"
        if _needs_refresh(existing, entry):
            repo.update(existing, _update_payload(entry))
            _logger.info("Refreshed built-in source %s", entry.config.key)
            return "refreshed"
        return "unchanged"


def _create_payload(entry: BuiltinSource):
    from academic_platform.infrastructure.database.schemas import (
        SearchSourceCreate,
    )

    config = entry.config
    return SearchSourceCreate(

        base_url=config.base_url,

        capabilities_json=json.dumps(config.capabilities),
        rate_limit_json=json.dumps(config.rate_limit) if config.rate_limit else None,
        auth_type=config.auth_type,
        credential_reference=config.credential_reference,
        enabled=config.enabled,
        timeout_seconds=config.timeout_seconds,
        result_limit=config.result_limit,
    )

    )

    config = entry.config
    return SearchSourceUpdate(
        display_name=config.display_name,
        base_url=config.base_url,
        endpoint=config.endpoint,
        http_method=config.http_method,
        parameter_map_json=json.dumps(config.parameter_map),
```

```
"""Application service for search source configuration.

The service is the single entry point the presentation layer uses to
list, enable, disable, test, and validate search sources. It is
constructed with a repository manager, a provider registry, and (when
available) the application's settings service so that it can honour the
user's HTTP and loopback preferences.

Design decisions
----------------
- The service never imports Qt. It emits plain data and raises typed
  exceptions; the presentation layer translates those to UI feedback.
- The service validates URLs before persisting them. Validation is
  fail-closed: a URL that fails validation is not persisted.

  work.

  ``SourceConfig``) rather than ORM instances. Callers never see ORM
  state.

from __future__ import annotations

import json
import logging
from collections.abc import Iterator
from contextlib import contextmanager
from dataclasses import dataclass
from typing import Any

from academic_platform.application.search_sources.errors import (

from academic_platform.application.publications.service import PublicationService


_logger = logging.getLogger(__name__)


@dataclass(frozen=True, slots=True)
class SourceValidationIssue:
    """A single validation issue for a source configuration."""

    field: str
    message: str


class SearchSourceService:
    """Application service for managing search sources."""

    def __init__(
        self,
        repositories: RepositoryManager,
        registry: ProviderRegistry,
        *,
        allow_http: bool = False,
        allow_loopback: bool = False,
    ) -> None:
        self._repositories = repositories
        self._registry = registry
        self._allow_http = allow_http
        self._allow_loopback = allow_loopback
        self._resolver = SourceResolver(registry)
        self._publications = PublicationService(repositories)

    @property
    def publications(self) -> PublicationService:
        """Return the publication service bound to the same repositories."""
        return self._publications

    @property
    def resolver(self) -> SourceResolver:
        """Return the source resolver used to construct provider adapters."""
        return self._resolver

    @contextmanager
    def _operation(self) -> Iterator[None]:
        """Wrap an operation in a transaction boundary.

        Repositories obtained through the manager are bound to a session
        that is opened on first access and closed when the operation
        completes. The manager is responsible for its own transaction
        boundary; the service does not commit or roll back explicitly.
        """
        try:
            yield
        finally:
            manager = self._repositories
            close_all = getattr(manager, "close_all", None)
            if callable(close_all):
                close_all()

    # ------------------------------------------------------------------
    # Reads
    # ------------------------------------------------------------------
    def list_sources(self) -> list[SourceRecord]:
        """Return every configured source ordered by display name."""
        with self._operation():
            repo = self._source_repository()
            rows = repo.list_all_ordered()
            return [self._to_record(row) for row in rows]

    def list_enabled(self) -> list[SourceRecord]:
        """Return enabled sources ordered by display name."""
        with self._operation():
            repo = self._source_repository()
            rows = repo.list_enabled()
            return [self._to_record(row) for row in rows]

    def get_by_key(self, key: str) -> SourceRecord:
        """Return the source with the given key.

        Raises
        ------
        SourceNotFoundError
            If no source matches the key.
        """
        with self._operation():
            repo = self._source_repository()
            row = repo.find_by_key(key)
            if row is None:
                raise SourceNotFoundError(f"No source with key {key!r}")
            return self._to_record(row)

    def get_by_provider_key(self, provider_key: str) -> SourceRecord | None:
        """Return the source with the given provider key, or ``None``."""
        with self._operation():
            repo = self._source_repository()
            row = repo.find_by_provider_key(provider_key)
            return self._to_record(row) if row is not None else None

    # ------------------------------------------------------------------
    # Validation
    # ------------------------------------------------------------------
    def validate(self, config: SourceConfig) -> list[SourceValidationIssue]:
        """Validate a source configuration without persisting it."""
        issues: list[SourceValidationIssue] = []
        if not config.key.strip():
            issues.append(SourceValidationIssue("key", "Key must not be empty"))
        if not config.provider_key.strip():
            issues.append(
                SourceValidationIssue("provider_key", "Provider key must not be empty")
            )
        if not config.display_name.strip():
            issues.append(
                SourceValidationIssue(
                    "display_name", "Display name must not be empty"
                )
            )
        if config.source_type not in {"builtin", "user_defined"}:
            issues.append(
                SourceValidationIssue(
                    "source_type",
                    "Source type must be 'builtin' or 'user_defined'",
                )
            )
        if config.timeout_seconds < 1 or config.timeout_seconds > 180:
            issues.append(
                SourceValidationIssue(
                    "timeout_seconds", "Timeout must be between 1 and 180 seconds"
                )
            )
        if config.result_limit < 1 or config.result_limit > 500:
            issues.append(
                SourceValidationIssue(
                    "result_limit", "Result limit must be between 1 and 500"
                )
            )
        if config.auth_type not in {"none", "api_key", "bearer_token"}:
            issues.append(
                SourceValidationIssue("auth_type", "Auth type is not supported")
            )
        if config.auth_type != "none" and not config.credential_reference:
            issues.append(
                SourceValidationIssue(
                    "credential_reference",
                    "A credential reference is required when authentication is enabled",
                )
            )
        url_result = validate_url(
            config.base_url,
            allow_http=self._allow_http,
            allow_loopback=self._allow_loopback,
        )
        if not url_result.ok:
            issues.append(SourceValidationIssue("base_url", url_result.message))
        if config.endpoint:
            endpoint_url = config.base_url.rstrip("/") + "/" + config.endpoint.lstrip("/")
            endpoint_result = validate_url(
                endpoint_url,
                allow_http=self._allow_http,
                allow_loopback=self._allow_loopback,
            )
            if not endpoint_result.ok:
                issues.append(
                    SourceValidationIssue("endpoint", endpoint_result.message)
                )
        return issues

    # ------------------------------------------------------------------
    # Writes
    # ------------------------------------------------------------------
    def create(self, config: SourceConfig) -> SourceRecord:
        """Persist a new source.

        Raises
        ------
        SourceValidationError
            If the configuration fails validation.
        """
        issues = self.validate(config)
        if issues:
            raise SourceValidationError([i.message for i in issues])
        with self._operation():
            repo = self._source_repository()
            payload = SearchSourceCreate(
                key=config.key,
                provider_key=config.provider_key,
                display_name=config.display_name,
                source_type=config.source_type,
                base_url=config.base_url,
                endpoint=config.endpoint,
                http_method=config.http_method,
                parameter_map_json=json.dumps(config.parameter_map),
                capabilities_json=json.dumps(config.capabilities),
                rate_limit_json=json.dumps(config.rate_limit)
                if config.rate_limit
                else None,
                auth_type=config.auth_type,
                credential_reference=config.credential_reference,
                enabled=config.enabled,
                timeout_seconds=config.timeout_seconds,
                result_limit=config.result_limit,
            )
            try:
                row = repo.create(payload)
            except DuplicateEntityError as exc:
                raise SourceValidationError(
                    [f"A source with key or provider key already exists: {exc}"]
                ) from exc
            self._resolver.invalidate(config.key)
            return self._to_record(row)

    def update(self, key: str, changes: dict[str, Any]) -> SourceRecord:
        """Apply changes to an existing source.

        Only the fields present in ``changes`` are modified. The
        resulting configuration is validated before persisting.
        """
        with self._operation():
            repo = self._source_repository()
            existing = repo.find_by_key(key)
            if existing is None:
                raise SourceNotFoundError(f"No source with key {key!r}")
            current = self._config_from_row(existing)
            merged = self._merge_config(current, changes)
            issues = self.validate(merged)
            if issues:
                raise SourceValidationError([i.message for i in issues])
            payload = SearchSourceUpdate(
                display_name=merged.display_name,
                base_url=merged.base_url,
                endpoint=merged.endpoint,
                http_method=merged.http_method,
                parameter_map_json=json.dumps(merged.parameter_map),
                capabilities_json=json.dumps(merged.capabilities),
                rate_limit_json=json.dumps(merged.rate_limit)
                if merged.rate_limit
                else None,
                auth_type=merged.auth_type,
                credential_reference=merged.credential_reference,
                enabled=merged.enabled,
                timeout_seconds=merged.timeout_seconds,
                result_limit=merged.result_limit,
            )
            try:
                row = repo.update(existing, payload)
            except DuplicateEntityError as exc:
                raise SourceValidationError([str(exc)]) from exc
            self._resolver.invalidate(key)
            return self._to_record(row)

    def set_enabled(self, key: str, enabled: bool) -> SourceRecord:
        """Enable or disable a source by key."""
        with self._operation():
            repo = self._source_repository()
            row = repo.find_by_key(key)
            if row is None:
                raise SourceNotFoundError(f"No source with key {key!r}")
            updated = repo.set_enabled(row.id, enabled)
            self._resolver.invalidate(key)
            return self._to_record(updated)

    def remove(self, key: str) -> None:
        """Remove a user-defined source.

        Built-in sources cannot be removed; they may only be disabled.
        Attempting to remove a built-in source raises
        ``SourceValidationError``.

        with self._operation():
            repo = self._source_repository()
            row = repo.find_by_key(key)
            if row is None:
                raise SourceNotFoundError(f"No source with key {key!r}")
            if row.source_type == "builtin":
                raise SourceValidationError(
                    [f"Built-in source {key!r} cannot be removed; disable it instead"]
                )
            try:
                repo.delete(row)
            except RepositoryError as exc:
                raise SourceValidationError([str(exc)]) from exc
            self._resolver.invalidate(key)
```

```
"""Resolve a persisted source configuration to a provider adapter.

The resolver bridges the application-layer ``SourceConfig`` and the
infrastructure-layer ``SearchProviderAdapter``. It reads the provider
registry, constructs the adapter with the source's configuration, and
caches the result for the lifetime of the resolver.

Design decisions
----------------
- The resolver is constructed per unit of work. Adapters are cached
  within the resolver, not across resolvers, so that configuration
  changes take effect on the next unit of work.
- The resolver never mutates the registry. It reads entries and
  constructs adapters.
- When a provider key is not registered, the resolver returns the null
  adapter from the registry rather than raising. This keeps the search
  page functional even when a source references a provider that the
  application does not yet ship.
"""

from __future__ import annotations

import logging

from academic_platform.application.search_sources.errors import (
    SourceValidationError,
)
from academic_platform.application.search_sources.source_config import SourceConfig
from academic_platform.infrastructure.search_providers.adapter import (
    SearchProviderAdapter,
)
from academic_platform.infrastructure.search_providers.errors import (
    ProviderConfigurationError,
)
from academic_platform.infrastructure.search_providers.registry import (
    ProviderRegistry,
)

_logger = logging.getLogger(__name__)


class SourceResolver:
    """Resolve source configurations to provider adapters."""

    def __init__(self, registry: ProviderRegistry) -> None:
        self._registry = registry
        self._cache: dict[str, SearchProviderAdapter] = {}

    def resolve(self, config: SourceConfig) -> SearchProviderAdapter:
        """Return the adapter for a source configuration.

        Raises
        ------
        SourceValidationError
            If the adapter cannot be constructed and no fallback is
            available. The resolver prefers to fall back to the null
            adapter when the provider is not registered; it raises only
            when the provider is registered but fails to construct with
            the supplied configuration.
        """
        cached = self._cache.get(config.key)
        if cached is not None:
            return cached

        provider_config = self._provider_config(config)
        try:
            adapter = self._registry.get(config.provider_key, config=provider_config)
        except ProviderConfigurationError as exc:
            if config.provider_key not in self._registry:
                _logger.info(
                    "Provider %r is not registered; falling back to the null adapter",
                    config.provider_key,
                )
                adapter = self._registry.get("null")
            else:
                raise SourceValidationError(
                    [f"Unable to construct provider {config.provider_key!r}: {exc}"]
                ) from exc

        self._cache[config.key] = adapter
        return adapter

    def invalidate(self, key: str | None = None) -> None:
        """Drop cached adapters.

        Passing ``None`` clears the entire cache; passing a source key
        drops only that entry. Callers use this after a source's
        configuration changes.
        """
        if key is None:
            self._cache.clear()
        else:
            self._cache.pop(key, None)

    @staticmethod
    def _provider_config(config: SourceConfig) -> dict[str, Any]:
        # The provider adapter receives the source's configuration as a
        # plain dictionary. Adapters interpret the keys they understand
        # and ignore the rest.
        return {
            "base_url": config.base_url,
            "endpoint": config.endpoint,
            "http_method": config.http_method,
            "parameter_map": dict(config.parameter_map),
            "capabilities": dict(config.capabilities),
            "rate_limit": dict(config.rate_limit) if config.rate_limit else None,
            "auth_type": config.auth_type,
            "credential_reference": config.credential_reference,
            "timeout_seconds": config.timeout_seconds,
            "result_limit": config.result_limit,
        }


from typing import Any  # noqa: E402  (needed by the type annotation above)
```

```
"""URL safety validator for search source configuration.

Every user-supplied URL is validated before it is stored in the database
and again before a request is issued. The validator implements the
network safeguards required by the product specification:

- Only supported schemes are permitted. ``https`` is the default; ``http``
  is permitted only when the caller explicitly allows it for a specific
  source (a local development exception).
- URLs with embedded credentials (``user:pass@host``) are rejected.
- Requests to loopback, private, link-local, and cloud metadata
  addresses are rejected unless the caller explicitly allows loopback for
  a local source.
- The resolved address is revalidated to reduce DNS rebinding risk.

Design decisions
----------------
- The validator returns a structured result rather than raising on
  every failure. Callers decide whether a failure is fatal or advisory.
- The validator never performs a network request. It uses the operating
  system resolver through ``socket.getaddrinfo``; the resolution is
  synchronous and is performed only when the caller asks for it.
- The list of blocked address ranges is defined explicitly rather than
  derived, so that adding a new range is a deliberate act.
"""

from __future__ import annotations

import ipaddress
import socket
from dataclasses import dataclass
from enum import Enum
from urllib.parse import urlsplit

_SUPPORTED_SCHEMES = ("https", "http")
_METADATA_ADDRESSES = (
    ipaddress.ip_network("169.254.169.254/32"),  # AWS, Azure, GCP
    ipaddress.ip_network("100.100.100.200/32"),  # Alibaba Cloud
    ipaddress.ip_network("fd00:ec2::254/128"),  # AWS IPv6 metadata
)


class UrlRejection(str, Enum):
    """Reason a URL was rejected."""

    EMPTY = "empty"
    MISSING_SCHEME = "missing_scheme"
    UNSUPPORTED_SCHEME = "unsupported_scheme"
    EMBEDDED_CREDENTIALS = "embedded_credentials"
    MISSING_HOST = "missing_host"
    INVALID_HOST = "invalid_host"
    LOOPBACK_NOT_ALLOWED = "loopback_not_allowed"
    PRIVATE_ADDRESS = "private_address"
    LINK_LOCAL_ADDRESS = "link_local_address"
    METADATA_ADDRESS = "metadata_address"
    RESOLUTION_FAILED = "resolution_failed"
    UNRESOLVED_HOST = "unresolved_host"


@dataclass(frozen=True, slots=True)
class UrlValidationResult:
    """Result of validating a URL."""

    url: str
    ok: bool
    rejection: UrlRejection | None = None
    message: str = ""
    host: str = ""
    scheme: str = ""
    resolved_addresses: tuple[str, ...] = ()

    @property
    def is_loopback(self) -> bool:
        """Return True when the URL resolves to a loopback address."""
        return any(
            ipaddress.ip_address(address).is_loopback
            for address in self.resolved_addresses
        )


def _is_ip_literal(host: str) -> tuple[bool, ipaddress.IPv4Address | ipaddress.IPv6Address | None]:
    try:
        return True, ipaddress.ip_address(host)
    except ValueError:
        return False, None


def _classify_ip(
    address: ipaddress.IPv4Address | ipaddress.IPv6Address,
) -> UrlRejection | None:
    for metadata_network in _METADATA_ADDRESSES:
        if address in metadata_network:
            return UrlRejection.METADATA_ADDRESS
    if address.is_loopback:
        return UrlRejection.LOOPBACK_NOT_ALLOWED
    if address.is_private and not address.is_loopback:
        return UrlRejection.PRIVATE_ADDRESS
    if address.is_link_local:
        return UrlRejection.LINK_LOCAL_ADDRESS
    return None


def _resolve_host(host: str, port: int | None) -> tuple[list[str], UrlRejection | None]:
    try:
        results = socket.getaddrinfo(
            host,
            port or 0,
            proto=socket.IPPROTO_TCP,
        )
    except socket.gaierror:
        return [], UrlRejection.RESOLUTION_FAILED
    addresses: list[str] = []
    for family, _socktype, _proto, _canonname, sockaddr in results:
        if family == socket.AF_INET:
            addresses.append(sockaddr[0])
        elif family == socket.AF_INET6:
            addresses.append(sockaddr[0])
    if not addresses:
        return [], UrlRejection.UNRESOLVED_HOST
    return addresses, None


def validate_url(
    url: str,
    *,
    allow_http: bool = False,
    allow_loopback: bool = False,
    resolve_host: bool = True,
) -> UrlValidationResult:
    """Validate a URL against the application's network safeguards.

    Parameters
    ----------
    url:
        The URL to validate.
    allow_http:
        When True, ``http`` is permitted in addition to ``https``. Used
        only for a specific source, never as a global policy.
    allow_loopback:
        When True, loopback addresses are permitted. Used only for a
        local source that the user explicitly configured.
    resolve_host:
        When True, the host is resolved and the resulting addresses are
        revalidated. When False, only the syntactic checks are performed.

    Returns
    -------
    UrlValidationResult
        The outcome. ``ok`` is True when the URL passes every check.
    """
    if not url or not url.strip():
        return UrlValidationResult(
            url=url,
            ok=False,
            rejection=UrlRejection.EMPTY,
            message="URL is empty",
        )
    stripped = url.strip()
    parts = urlsplit(stripped)
    if not parts.scheme:
        return UrlValidationResult(
            url=stripped,
            ok=False,
            rejection=UrlRejection.MISSING_SCHEME,
            message="URL must include a scheme (https://)",
        )
    scheme = parts.scheme.lower()
    if scheme not in _SUPPORTED_SCHEMES:
        return UrlValidationResult(
            url=stripped,
            ok=False,
            rejection=UrlRejection.UNSUPPORTED_SCHEME,
            message=f"Scheme {scheme!r} is not supported",
            scheme=scheme,
        )
    if scheme == "http" and not allow_http:
        return UrlValidationResult(
            url=stripped,
            ok=False,
            rejection=UrlRejection.UNSUPPORTED_SCHEME,
            message=(
                "HTTP is not permitted. Use HTTPS, or explicitly allow HTTP "
                "for this specific source."
            ),
            scheme=scheme,
        )
    if parts.username or parts.password:
        return UrlValidationResult(
            url=stripped,
            ok=False,
            rejection=UrlRejection.EMBEDDED_CREDENTIALS,
            message="Credentials embedded in URLs are not permitted",
            scheme=scheme,
        )
    host = parts.hostname or ""
    if not host:
        return UrlValidationResult(
            url=stripped,
            ok=False,
            rejection=UrlRejection.MISSING_HOST,
            message="URL must include a host",
            scheme=scheme,
        )

    is_ip, ip = _is_ip_literal(host)
    if is_ip and ip is not None:
        rejection = _classify_ip(ip)
        if rejection is UrlRejection.LOOPBACK_NOT_ALLOWED and allow_loopback:
            rejection = None
        if rejection is not None:
            return UrlValidationResult(
                url=stripped,
                ok=False,
                rejection=rejection,
                message=_rejection_message(rejection),
                scheme=scheme,
                host=host,
                resolved_addresses=(str(ip),),
            )
        return UrlValidationResult(
            url=stripped,
            ok=True,
            scheme=scheme,
            host=host,
            resolved_addresses=(str(ip),),
        )

    if not resolve_host:
        return UrlValidationResult(
            url=stripped,
            ok=True,
            scheme=scheme,
            host=host,
        )

    port = parts.port
    addresses, resolution_error = _resolve_host(host, port)
    if resolution_error is not None:
        return UrlValidationResult(
            url=stripped,
            ok=False,
            rejection=resolution_error,
            message=_rejection_message(resolution_error),
            scheme=scheme,
            host=host,
        )

    for address_str in addresses:
        try:
            address = ipaddress.ip_address(address_str)
        except ValueError:
            continue
        rejection = _classify_ip(address)
        if rejection is UrlRejection.LOOPBACK_NOT_ALLOWED and allow_loopback:
            continue
        if rejection is not None:
            return UrlValidationResult(
                url=stripped,
                ok=False,
                rejection=rejection,
                message=_rejection_message(rejection),
                scheme=scheme,
                host=host,
                resolved_addresses=tuple(addresses),
            )
    return UrlValidationResult(
        url=stripped,
        ok=True,
        scheme=scheme,
        host=host,
        resolved_addresses=tuple(addresses),
    )


def _rejection_message(rejection: UrlRejection) -> str:
    mapping = {
        UrlRejection.EMPTY: "URL is empty",
        UrlRejection.MISSING_SCHEME: "URL must include a scheme (https://)",
        UrlRejection.UNSUPPORTED_SCHEME: "Scheme is not supported",
        UrlRejection.EMBEDDED_CREDENTIALS: (
            "Credentials embedded in URLs are not permitted"
        ),
        UrlRejection.MISSING_HOST: "URL must include a host",
        UrlRejection.INVALID_HOST: "URL host is not valid",
        UrlRejection.LOOPBACK_NOT_ALLOWED: (
            "Loopback addresses are not permitted for this source"
        ),
        UrlRejection.PRIVATE_ADDRESS: (
            "Private addresses are not permitted"
        ),
        UrlRejection.LINK_LOCAL_ADDRESS: (
            "Link-local addresses are not permitted"
        ),
        UrlRejection.METADATA_ADDRESS: (
            "Cloud metadata addresses are not permitted"
        ),
        UrlRejection.RESOLUTION_FAILED: (
            "The host could not be resolved"
        ),
        UrlRejection.UNRESOLVED_HOST: (
            "The host did not resolve to any address"
        ),
    }
    return mapping.get(rejection, "URL is not valid")
```

```
    def _ensure_schema_at_head(self) -> None:
        """Ensure the database schema is at the current head revision.

        The bootstrap normally applies migrations before constructing the
        window. Tests and other callers may construct the window against
        a freshly created database whose schema has not been applied.
        Running the migration here is idempotent and cheap when the
        schema is already at head. Failures are logged and tolerated: the
        window remains constructible and individual pages report their
        own errors when they attempt to read.
        """
        try:
            from academic_platform.infrastructure.database.migrations import (
                upgrade_to_head,
            )

            url = self._database.configuration.url
            upgrade_to_head(url)
        except Exception:  # noqa: BLE001 - defensive boundary
            _logger.exception(
                "Unable to ensure the database schema is at head"
            )

        quit_action.triggered.connect(self.close)
        file_menu.addAction(quit_action)

            action.triggered.connect(

            view_menu.addAction(action)

    def _register_pages(self) -> None:
        for destination in DESTINATIONS:
            page = create_page(
                destination=destination,
                directories=self._directories,
                settings_service=self._settings_service,
                database=self._database,
                parent=self,
            )
            self._pages[destination.identifier] = page
            index = self._stack.addWidget(page)
            self._page_indices[destination.identifier] = index
        self._wire_library_page()
        self._wire_projects_page()
        self._wire_collections_page()
        self._wire_research_assistant_page()


    def _wire_library_page(self) -> None:
        """Wire the library page to the application services and the import dialog."""
        library_page = self._pages.get("library")
        if not isinstance(library_page, LibraryPage):
            return
        # The service is constructed per call so that each operation runs
        # in its own transaction boundary.
        try:
            service = self._library_service_factory()
        except Exception:  # noqa: BLE001 - boundary
            _logger.exception("Unable to construct the library service")
            return
        library_page.attach_service(service)
        library_page.attach_import_factory(self._open_import_dialog)
        library_page.open_document_requested.connect(self._on_open_document_requested)
        library_page.add_to_collection_requested.connect(
            self._on_add_to_collection_requested
        )
        library_page.add_to_project_requested.connect(
            self._on_add_to_project_requested
        )

    def _open_import_dialog(self, parent) -> None:
        """Open the import dialog with the application's coordinator factory."""
        notification_service = self._notification_service

        def _on_imported(_result) -> None:
            library_page = self._pages.get("library")
            if isinstance(library_page, LibraryPage):
                library_page.refresh()

        open_import_dialog(
            parent,
            self._import_coordinator_factory,
            notification_service=notification_service,
            on_imported=_on_imported,
        )

    def _on_open_document_requested(self, document_key: str) -> None:
        """Handle a document-open request from the library page.

        The document reader is delivered in a later phase. The handler
        publishes an informational notification so the user receives
        immediate feedback.
        """
        if self._notification_service is not None:
            self._notification_service.info(
                "Open document",
                f"The reader is not yet available ({document_key}).",
            )


    def _connect_signals(self) -> None:
        self._navigation.destination_selected.connect(self._on_destination_selected)
        from academic_platform.ui.pages.dashboard_page import DashboardPage
        from academic_platform.ui.pages.research_assistant import (
            ResearchAssistantPage,
        )
        from academic_platform.ui.pages.search import GlobalSearchPage

        dashboard = self._pages.get("dashboard")
        if isinstance(dashboard, DashboardPage):
            dashboard.destination_requested.connect(self.navigate_to)

        discover = self._pages.get("discover")
        if isinstance(discover, GlobalSearchPage):
            discover.search_submitted.connect(self._on_search_submitted)
            discover.retry_requested.connect(self._on_search_retry_requested)

        assistant = self._pages.get("research_assistant")
        if isinstance(assistant, ResearchAssistantPage):
            assistant.destination_requested.connect(self.navigate_to)

    def _install_shortcuts(self) -> None:
        """Install global navigation shortcuts for every destination."""
        for destination in DESTINATIONS:
            if not destination.shortcut:
                continue

            shortcut.setContext(Qt.ShortcutContext.ApplicationShortcut)
            shortcut.activated.connect(
                lambda key=destination.identifier: self.navigate_to(key)
            )
            self._shortcuts.append(shortcut)

    def navigate_to(self, identifier: str) -> None:
        """Navigate to the destination with the given identifier.

        Unknown identifiers are logged and ignored; the current page remains
        active. Known identifiers cause the previous page to be deactivated
        and the new page to be activated.
        """
        page = self._pages.get(identifier)
        if page is None:
            _logger.warning("Unknown destination requested: %s", identifier)
            return

        if self._current_identifier == identifier:
            page.on_activated()
            return

        previous = self._pages.get(self._current_identifier) if self._current_identifier else None
        if previous is not None:
            previous.on_deactivated()

        self._stack.setCurrentWidget(page)
        self._current_identifier = identifier
        page.on_activated()
        self.statusBar().showMessage(page.title)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    def refresh_active_user(self) -> str | None:
        """Read the active user key from the database and cache it.

        The value is refreshed after a user-visible settings change and
        once during window construction. When the settings tables are not
        yet present (for example in tests that construct the window
        against an unmigrated database), the method treats the active
        user as ``None`` and logs the reason at debug level.

        Returns
        -------
        str | None
            The active user key, or ``None`` when no user is active or the
            database cannot answer yet.
        """
        try:
            with unit_of_work(self._database) as repositories:
                bridge = SettingsPersistenceBridge(
                    settings_service=self._settings_service,
                    app_settings_repository=repositories.get("app_settings"),
                    user_profile_repository=repositories.get("user_profile"),
                )
                key = bridge.get_active_user_key()
        except SettingsPersistenceError:
            _logger.exception("Unable to read active user key")
            return self._active_user_key
        except Exception as exc:  # noqa: BLE001 - defensive boundary
            _logger.debug("Active user key unavailable: %s", exc)
            return self._active_user_key
        self._active_user_key = key
        return key

class _PerCallManager:
    """A repository manager that opens a fresh session per call.

    Each call to :meth:`get` opens a session, wraps the repository so that
    the session is committed on success, rolled back on failure, and
    closed in all cases before the call returns. This prevents SQLite
    from holding the file lock across the multiple repository accesses
    that make up a single search.
    """

    def __init__(self, database) -> None:
        self._database = database
        self._sessions: list = []

    def get(self, key: str):
        from academic_platform.infrastructure.repositories.factory import (
            _REPOSITORY_REGISTRY,
        )

        session = self._database._session_factory()  # noqa: SLF001
        self._sessions.append(session)
        repository_class = _REPOSITORY_REGISTRY.get(key)
        if repository_class is None:
            from academic_platform.infrastructure.repositories.errors import (
                RepositoryError,
            )

            session.close()
            raise RepositoryError(f"No repository registered under key {key!r}")
        return _SessionClosingRepository(repository_class(session), session)

    def close_all(self) -> None:
        for session in self._sessions:
            try:
                session.commit()
            except Exception:  # noqa: BLE001
                session.rollback()
            finally:
                session.close()
        self._sessions.clear()


class _SessionClosingRepository:
    """Proxy that commits and closes the session after each repository call."""

    def __init__(self, repository, session) -> None:
        self._repository = repository
        self._session = session

    def __getattr__(self, name):
        target = getattr(self._repository, name)
        if not callable(target):
            return target

        def _call(*args, **kwargs):
            try:
                result = target(*args, **kwargs)
                self._session.commit()
                return result
            except Exception:
                try:
                    self._session.rollback()
                except Exception:  # noqa: BLE001
                    pass
                raise
            finally:
                try:
                    self._session.close()
                except Exception:  # noqa: BLE001
                    pass

        return _call
```
